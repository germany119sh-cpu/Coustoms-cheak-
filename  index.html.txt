<script>
    // دیتابیس فرضی (می‌توانید کدهای واقعی را اینجا اضافه کنید)
    const database = [
        { name: "روغن", tsc: "1234", hs: "1512", country: "مالزی", tax: "5%", note: "کالای اساسی" },
        { name: "تایر", tsc: "5678", hs: "4011", country: "چین", tax: "20%", note: "مجوز وزارت صنایع" },
        { name: "لپ تاپ", tsc: "9900", hs: "8471", country: "دبی", tax: "10%", note: "تجهیزات الکترونیک" }
    ];

    function searchAction() {
        const name = document.getElementById('nameInput').value;
        const resultDiv = document.getElementById('result');
        
        // پیدا کردن جنس بر اساس نام (جستجوی ساده)
        const found = database.find(item => item.name.includes(name) && name !== "");

        if (found) {
            resultDiv.style.display = "block";
            resultDiv.innerHTML = `
                <div class="item-row"><span class="label">نام:</span> <span>${found.name}</span></div>
                <div class="item-row"><span class="label">HS کد:</span> <span>${found.hs}</span></div>
                <div class="item-row"><span class="label">TSC کد:</span> <span>${found.tsc}</span></div>
                <div class="item-row"><span class="label">کشور:</span> <span>${found.country}</span></div>
                <div class="item-row"><span class="label">تعرفه گمرکی:</span> <span style="color:#00ff88">${found.tax}</span></div>
                <div class="item-row"><span class="label">ملاحظات:</span> <span>${found.note}</span></div>
            `;
        } else {
            resultDiv.style.display = "block";
            resultDiv.innerHTML = "❌ محصولی با این مشخصات یافت نشد.";
        }
    }
</script>
